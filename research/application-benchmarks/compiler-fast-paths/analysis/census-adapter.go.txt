// A defined type adds the benchmark's methods; arithmetic uses the fork's
// operators directly, with no dynamic dispatch or second decimal engine.
type Decimal128 decimal128
type Rounding decimal.RoundingMode

const (
	TiesToEven     = Rounding(decimal.ToNearestEven)
	TiesToAway     = Rounding(decimal.ToNearestAway)
	TowardZero     = Rounding(decimal.ToZero)
	TowardPositive = Rounding(decimal.ToPositiveInf)
	TowardNegative = Rounding(decimal.ToNegativeInf)
)

func FromBID128(hi, lo uint64) Decimal128 { return Decimal128(decimal.BID128frombits(hi, lo)) }
func FromInt64(n int64) Decimal128        { return Decimal128(n) }
func Zero() Decimal128                    { return 0 }
func MustParse(s string) Decimal128 {
	x, err := strconv.ParseDecimal(s, 128)
	if err != nil {
		panic(err)
	}
	return Decimal128(x)
}

// GDA uses fixed notation whenever quantum <= 1 and adjusted exponent >= -6,
// including values >= 1e21 where strconv's canonical form is scientific.
// It also spells infinities in full and exponent digits without zero padding.
func (x Decimal128) String() string { census["format.calls"]++;
	// The canonical form fits a sign, 0., five leading zeros and all
	// coefficient digits. Adapt in that buffer, then allocate one string.
	var buf [len("-0.00000") + decimal128Digits]byte
	s := strconv.AppendDecimal(buf[:0], decimal128(x), 'G', -1, 128)
	if len(s) == len("+Inf") && s[1] == 'I' {
		if s[0] == '-' {
			return "-Infinity"
		}
		return "Infinity"
	}
	if i := bytes.IndexByte(s, 'E'); i >= 0 {
		j := i + 2 // exponent digits, after its sign
		for j < len(s)-1 && s[j] == '0' {
			j++
		}
		if s[i+1] == '+' {
			exp := 0
			for _, c := range s[j:] {
				exp = exp*10 + int(c-'0')
			}
			start := 0
			if s[0] == '-' {
				start++
			}
			digits := i - start
			if s[start+1] == '.' {
				digits--
			}
			// Quantum exponent = exp - (digits-1). A non-positive
			// quantum on this positive-exponent form needs fixed notation.
			if exp < digits {
				copy(s[start+1:], s[start+2:i]) // remove the old point
				end := i - 1
				point := start + exp + 1
				if point < end {
					copy(s[point+1:], s[point:end])
					s[point] = '.'
					end++
				}
				return string(s[:end])
			}
		}
		n := copy(s[i+2:], s[j:]) // remove exponent padding in place
		s = s[:i+2+n]
	}
	return string(s)
}

func (x Decimal128) Add(y Decimal128) Decimal128      { censusRecord("add",x,y); return x + y }
func (x Decimal128) Sub(y Decimal128) Decimal128      { censusRecord("sub",x,y); return x - y }
func (x Decimal128) Mul(y Decimal128) Decimal128      { censusRecord("mul",x,y); return x * y }
func (x Decimal128) Quo(y Decimal128) Decimal128      { censusRecord("div",x,y); return x / y }
func (x Decimal128) Neg() Decimal128                  { return -x }
func (x Decimal128) Quantize(q Decimal128) Decimal128 { return decimal.Quantize(x, q) }
func (x Decimal128) Less(y Decimal128) bool           { censusRecord("less",x,y); return x < y }
func (x Decimal128) Equal(y Decimal128) bool          { censusRecord("equal",x,y); return x == y }
func (x Decimal128) Signbit() bool                    { return decimal.Signbit(x) }
func (x Decimal128) IsZero() bool                     { censusRecord("isZero",x,0); return x == 0 }
func (x Decimal128) IsNaN() bool                      { return decimal.IsNaN(x) }
func (x Decimal128) IsInf() bool                      { return decimal.IsInf(x, 0) }
func Minimum(x, y Decimal128) Decimal128              { return min(x, y) }
func Maximum(x, y Decimal128) Decimal128              { return max(x, y) }

func (r Rounding) RoundToIntegral(x Decimal128) Decimal128 {
	switch r {
	case TiesToEven:
		return decimal.RoundToEven(x)
	case TiesToAway:
		return decimal.Round(x)
	case TowardZero:
		return decimal.Trunc(x)
	case TowardPositive:
		return decimal.Ceil(x)
	case TowardNegative:
		return decimal.Floor(x)
	default:
		panic("invalid rounding direction")
	}
}
func (r Rounding) ToInt64(x Decimal128) int64 { return int64(r.RoundToIntegral(x)) }
func (r Rounding) Quantize(x, q Decimal128) Decimal128 {
	return decimal.QuantizeMode(x, q, decimal.RoundingMode(r))
}
