// tax_preparation CA harness on the d128 backend (decimal128-go).
package main

import (
	"os"
 "runtime/pprof"

	"github.com/migueldecimal128/go-decimal128-app-benchmark/apps/tax_preparation/internal/tax"
)

func main() {
 f,err:=os.Create(os.Getenv("DECIMAL_DIAGNOSTIC_PROFILE"));if err!=nil {panic(err)}
 if err=pprof.StartCPUProfile(f);err!=nil {panic(err)}
 code:=tax.RunCA()
 pprof.StopCPUProfile();if err=f.Close();err!=nil {panic(err)}
 os.Exit(code)
}
